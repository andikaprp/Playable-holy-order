templar_senior_knight_validity_trigger = {
	base_court_position_validity_trigger = { EMPLOYER = $EMPLOYER$ }
	is_capable_adult = yes
	is_imprisoned = no
	faith = $EMPLOYER$.faith
	NOT = { has_trait = excommunicated }
	NOT = { is_spouse_of = $EMPLOYER$ }
	OR = {
		is_courtier_of = $EMPLOYER$
		AND = {
			is_vassal_of = $EMPLOYER$
			any_held_title = { has_variable = templar_chapter_voice }
		}
	}
	can_be_combatant_based_on_gender_trigger = { ARMY_OWNER = $EMPLOYER$ }
	custom_tooltip = {
		text = none_does_not_have_a_full_time_job
		trigger_if = {
			limit = { NOT = { has_court_position = templar_senior_knight_court_position } }
			court_position_does_not_already_have_a_job_trigger = yes
		}
	}
	OR = {
		is_landed = no
		any_held_title = { has_variable = templar_chapter_voice }
	}
	OR = {
		has_trait = order_member
		is_knight_of = $EMPLOYER$
		martial >= 6
		prowess >= 8
	}
}

is_landless_holy_order = {

	government_has_flag = government_is_holy_order

	is_ruler = yes

	any_held_title = { is_holy_order = yes }

	NOT = {

		any_held_title = {
			is_noble_family_title = no
			NOT = { is_holy_order = yes }
			NOT = { has_variable = templar_chapter_domicile_anchor }
			NOT = { has_variable = templar_chapter_voice }
		}

	}

}



is_playable_holy_order = {

	government_has_flag = government_is_holy_order

	is_ruler = yes

	OR = {

		is_landless_holy_order = yes

		has_character_flag = templar_playable_grandmaster

	}

}



templar_grandmaster_setup_incomplete_trigger = {

	OR = {

		NOT = { government_has_flag = government_is_holy_order }

		any_held_title = {

			NOT = { is_holy_order = yes }
			NOT = { has_variable = templar_chapter_domicile_anchor }
			NOT = { has_variable = templar_chapter_voice }
		}

		AND = {

			has_dlc_feature = roads_to_power

			NOT = { has_domicile = yes }

		}

	}

}

	tier = tier_barony

	NOT = { has_variable = templar_commandery_order }

	NOT = { is_leased_out = yes }

}
