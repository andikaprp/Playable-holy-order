templar_play_as_grandmaster_interaction = {
	common_interaction = yes
	category = interaction_category_religion
	interface_priority = 55
	icon = holy_order
	desc = templar_play_as_grandmaster_interaction_desc
	notification_text = templar_play_as_grandmaster_interaction_notification
	auto_accept = yes
	force_notification = yes
	use_diplomatic_range = no

	is_shown = {
		scope:actor = { is_ai = no }
		scope:recipient = {
			government_has_flag = government_is_holy_order
			is_alive = yes
			is_ruler = yes
			primary_title = { is_holy_order = yes }
		}
		scope:recipient != scope:actor
	}

	is_valid = {
		scope:recipient = { is_imprisoned = no }
	}

	on_accept = {
		hidden_effect = {
			templar_play_as_grandmaster_effect = yes
		}
	}
}
