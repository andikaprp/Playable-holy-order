templar_defend_commanderies_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_realm.dds" }
	desc = templar_defend_commanderies_decision_desc
	selection_tooltip = templar_defend_commanderies_decision_tooltip
	cooldown = { years = 3 }
	ai_check_interval = 0
	is_shown = {
		has_variable = templar_simulated_order
		has_variable_list = templar_commanderies_list
	}
	is_valid = {
		any_in_list = {
			variable = templar_commanderies_list
			var:templar_commandery_patron ?= {
				is_at_war = yes
				any_war_enemy = {
					faith = {
						faith_hostility_level = {
							target = root.faith
							value >= faith_hostile_level
						}
					}
				}
			}
		}
	}
	effect = {
		add_character_modifier = { modifier = templar_defending_commanderies_modifier years = 3 }
		every_in_list = {
			variable = templar_commanderies_list
			if = {
				limit = { var:templar_commandery_patron ?= { is_at_war = yes } }
				change_variable = { name = templar_commandery_loyalty add = 10 }
				if = { limit = { var:templar_commandery_loyalty > 100 } set_variable = { name = templar_commandery_loyalty value = 100 } }
			}
		}
		add_piety = 200
		templar_change_reputation_effect = { VALUE = 6 }
		templar_rebuild_commandery_bonuses_effect = yes
	}
	ai_potential = { always = no }
}
