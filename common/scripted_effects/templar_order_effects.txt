templar_initialize_order_effect = {
	set_variable = { name = templar_discipline value = 55 }
	set_variable = { name = templar_zeal value = 60 }
	set_variable = { name = templar_reputation value = 20 }
	set_variable = { name = templar_commanderies value = 0 }
	set_variable = { name = templar_commandery_limit value = 5 }
	set_variable = { name = templar_rank value = 1 }
	templar_update_order_effect = yes
}

templar_normalize_resources_effect = {
	if = { limit = { var:templar_discipline > 100 } set_variable = { name = templar_discipline value = 100 } }
	if = { limit = { var:templar_discipline < 0 } set_variable = { name = templar_discipline value = 0 } }
	if = { limit = { var:templar_zeal > 100 } set_variable = { name = templar_zeal value = 100 } }
	if = { limit = { var:templar_zeal < 0 } set_variable = { name = templar_zeal value = 0 } }
	if = { limit = { var:templar_reputation > 100 } set_variable = { name = templar_reputation value = 100 } }
	if = { limit = { var:templar_reputation < 0 } set_variable = { name = templar_reputation value = 0 } }
}

templar_update_status_effect = {
	remove_character_modifier = templar_high_discipline_modifier
	remove_character_modifier = templar_low_discipline_modifier
	remove_character_modifier = templar_high_zeal_modifier
	remove_character_modifier = templar_low_zeal_modifier
	remove_character_modifier = templar_high_reputation_modifier
	remove_character_modifier = templar_low_reputation_modifier

	if = {
		limit = { var:templar_discipline >= 70 }
		add_character_modifier = { modifier = templar_high_discipline_modifier }
	}
	else_if = {
		limit = { var:templar_discipline <= 30 }
		add_character_modifier = { modifier = templar_low_discipline_modifier }
	}
	if = {
		limit = { var:templar_zeal >= 70 }
		add_character_modifier = { modifier = templar_high_zeal_modifier }
	}
	else_if = {
		limit = { var:templar_zeal <= 30 }
		add_character_modifier = { modifier = templar_low_zeal_modifier }
	}
	if = {
		limit = { var:templar_reputation >= 70 }
		add_character_modifier = { modifier = templar_high_reputation_modifier }
	}
	else_if = {
		limit = { var:templar_reputation <= 30 }
		add_character_modifier = { modifier = templar_low_reputation_modifier }
	}
}

templar_update_rank_effect = {
	remove_character_modifier = templar_rank_regional_modifier
	remove_character_modifier = templar_rank_renowned_modifier
	remove_character_modifier = templar_rank_guardian_modifier
	remove_character_modifier = templar_rank_legendary_modifier
	remove_character_modifier = templar_rank_shield_modifier

	if = {
		limit = { var:templar_reputation >= 90 }
		set_variable = { name = templar_rank value = 5 }
		add_character_modifier = { modifier = templar_rank_shield_modifier }
	}
	else_if = {
		limit = { var:templar_reputation >= 70 }
		set_variable = { name = templar_rank value = 4 }
		add_character_modifier = { modifier = templar_rank_legendary_modifier }
	}
	else_if = {
		limit = { var:templar_reputation >= 50 }
		set_variable = { name = templar_rank value = 3 }
		add_character_modifier = { modifier = templar_rank_guardian_modifier }
	}
	else_if = {
		limit = { var:templar_reputation >= 30 }
		set_variable = { name = templar_rank value = 2 }
		add_character_modifier = { modifier = templar_rank_renowned_modifier }
	}
	else = {
		set_variable = { name = templar_rank value = 1 }
		add_character_modifier = { modifier = templar_rank_regional_modifier }
	}
}

	templar_normalize_resources_effect = yes
	templar_update_status_effect = yes
	templar_update_rank_effect = yes

templar_update_commandery_limit_effect = {
	set_variable = { name = templar_commandery_limit value = 4 }
	change_variable = { name = templar_commandery_limit add = var:templar_rank }
	if = {
		limit = { piety_level >= 3 }
		change_variable = { name = templar_commandery_limit add = 1 }
	}
	if = {
		limit = { has_character_modifier = templar_endorsed_modifier }
		change_variable = { name = templar_commandery_limit add = 1 }
	}
	if = {
		limit = { var:templar_commandery_limit > 12 }
		set_variable = { name = templar_commandery_limit value = 12 }
	}
}

templar_rebuild_commandery_bonuses_effect = {
	remove_character_modifier = templar_commandery_small_modifier
	remove_character_modifier = templar_commandery_medium_modifier
	remove_character_modifier = templar_commandery_headquarters_modifier
	remove_character_modifier = templar_commandery_loyal_modifier
	remove_character_modifier = templar_commandery_disloyal_modifier
	remove_character_modifier = templar_upgrade_barracks_modifier
	remove_character_modifier = templar_upgrade_stables_modifier
	remove_character_modifier = templar_upgrade_hospice_modifier
	remove_character_modifier = templar_upgrade_chapel_modifier
	remove_character_modifier = templar_upgrade_armory_modifier
	remove_character_modifier = templar_upgrade_library_modifier
	remove_character_modifier = templar_commandery_operating_rights_modifier
	set_variable = { name = templar_commanderies value = 0 }

	if = {
		limit = { has_variable_list = templar_commanderies_list }
		every_in_list = {
			variable = templar_commanderies_list
			limit = { var:templar_commandery_order ?= root }
			root = {
				change_variable = { name = templar_commanderies add = 1 }
				add_character_modifier = { modifier = templar_commandery_operating_rights_modifier }
			}
			if = {
				limit = { var:templar_commandery_level >= 3 }
				root = { add_character_modifier = { modifier = templar_commandery_headquarters_modifier } }
			}
			else_if = {
				limit = { var:templar_commandery_level >= 2 }
				root = { add_character_modifier = { modifier = templar_commandery_medium_modifier } }
			}
			else = {
				root = { add_character_modifier = { modifier = templar_commandery_small_modifier } }
			}
			if = {
				limit = { var:templar_commandery_loyalty >= 70 }
				root = { add_character_modifier = { modifier = templar_commandery_loyal_modifier } }
			}
			else_if = {
				limit = { var:templar_commandery_loyalty <= 30 }
				root = { add_character_modifier = { modifier = templar_commandery_disloyal_modifier } }
			}
			if = { limit = { has_variable = templar_upgrade_barracks } root = { add_character_modifier = { modifier = templar_upgrade_barracks_modifier } } }
			if = { limit = { has_variable = templar_upgrade_stables } root = { add_character_modifier = { modifier = templar_upgrade_stables_modifier } } }
			if = { limit = { has_variable = templar_upgrade_hospice } root = { add_character_modifier = { modifier = templar_upgrade_hospice_modifier } } }
			if = { limit = { has_variable = templar_upgrade_chapel } root = { add_character_modifier = { modifier = templar_upgrade_chapel_modifier } } }
			if = { limit = { has_variable = templar_upgrade_armory } root = { add_character_modifier = { modifier = templar_upgrade_armory_modifier } } }
			if = { limit = { has_variable = templar_upgrade_library } root = { add_character_modifier = { modifier = templar_upgrade_library_modifier } } }
		}
	}
	templar_update_commandery_limit_effect = yes
}

templar_update_commandery_level_effect = {
	if = {
		limit = { var:templar_commandery_investment >= 4 }
		set_variable = { name = templar_commandery_level value = 3 }
		title_province.county = {
			remove_county_modifier = templar_commandery_small_county_modifier
			remove_county_modifier = templar_commandery_medium_county_modifier
			remove_county_modifier = templar_commandery_headquarters_county_modifier
			add_county_modifier = { modifier = templar_commandery_headquarters_county_modifier }
		}
	}
	else_if = {
		limit = { var:templar_commandery_investment >= 2 }
		set_variable = { name = templar_commandery_level value = 2 }
		title_province.county = {
			remove_county_modifier = templar_commandery_small_county_modifier
			remove_county_modifier = templar_commandery_medium_county_modifier
			remove_county_modifier = templar_commandery_headquarters_county_modifier
			add_county_modifier = { modifier = templar_commandery_medium_county_modifier }
		}
	}
	else = {
		set_variable = { name = templar_commandery_level value = 1 }
		title_province.county = {
			remove_county_modifier = templar_commandery_small_county_modifier
			remove_county_modifier = templar_commandery_medium_county_modifier
			remove_county_modifier = templar_commandery_headquarters_county_modifier
			add_county_modifier = { modifier = templar_commandery_small_county_modifier }
		}
	}
}

	set_variable = templar_temple_bank_active
	add_trait = templar_banker
	add_character_modifier = { modifier = templar_bank_of_the_temple_modifier }
	add_to_global_variable_list = { name = templar_active_banks target = root }
}

templar_update_bank_tier_effect = {
	if = {
		limit = { has_variable = templar_temple_bank_active }
		if = {
			limit = { var:templar_rank >= 5 has_trait = templar_deposit_banker }
			remove_trait = templar_deposit_banker
			add_trait = templar_insurer
		}
		else_if = {
			limit = {
				var:templar_rank >= 3
				NOT = { has_trait = templar_deposit_banker }
				NOT = { has_trait = templar_insurer }
			}
			remove_trait = templar_banker
			add_trait = templar_deposit_banker
		}
	}
}

templar_grant_loan_effect = {
	$LENDER$ = {
		add_gold = { value = $AMOUNT$ multiply = -1 }
		templar_change_reputation_effect = { VALUE = 2 }
		add_to_variable_list = { name = templar_bank_clients target = $BORROWER$ }
		add_to_variable_list = { name = owes_me_money target = $BORROWER$ }
	}
	$BORROWER$ = {
		add_gold = $AMOUNT$
		add_trait = templar_bank_client_trait
		set_variable = { name = templar_loan_amount_owed value = $AMOUNT$ }
		set_variable = { name = templar_loan_years_elapsed value = 0 }
		set_variable = { name = templar_bank_client_of value = $LENDER$ }
		add_opinion = { target = $LENDER$ modifier = templar_owed_favor_opinion opinion = 10 }
	}
}

templar_clear_loan_effect = {
	if = {
		limit = { exists = var:templar_bank_client_of }
		var:templar_bank_client_of = {
			remove_list_variable = { name = owes_me_money target = prev }
		}
	}
	remove_variable = templar_loan_amount_owed
	remove_variable = templar_loan_years_elapsed
	remove_variable = templar_loan_in_default
	remove_variable = templar_bank_client_of
}

templar_make_deposit_effect = {
	$BANK$ = {
		add_gold = $NET$
		templar_change_reputation_effect = { VALUE = 3 }
		add_to_variable_list = { name = templar_bank_clients target = $DEPOSITOR$ }
		add_to_variable_list = { name = templar_depositors target = $DEPOSITOR$ }
	}
	$DEPOSITOR$ = {
		add_gold = -$GROSS$
		add_trait = templar_bank_client_trait
		set_variable = { name = templar_deposit_amount value = $NET$ }
		set_variable = { name = templar_bank_client_of value = $BANK$ }
		add_opinion = { target = $BANK$ modifier = templar_grateful_client_opinion opinion = 15 }
	}
}

templar_withdraw_deposit_effect = {
	$DEPOSITOR$ = {
		if = {
			limit = { exists = var:templar_bank_client_of }
			var:templar_bank_client_of = {
				add_gold = { value = $NET$ multiply = -1 }
				remove_list_variable = { name = templar_depositors target = prev }
			}
			add_gold = $NET$
		}
		remove_variable = templar_deposit_amount
		remove_variable = templar_bank_client_of
	}
}

set_up_templar_chapter_domicile_effect = {
	if = {
		limit = { government_has_flag = government_is_holy_order }
		domicile = {
			if = {
				limit = { NOT = { has_domicile_building_or_higher = templar_chapter_main_01 } }
				add_domicile_building = templar_chapter_main_01
			}
		}
	}
}

templar_ensure_playable_grandmaster_effect = {
	add_character_flag = templar_playable_grandmaster
	if = {
		limit = { NOT = { has_government = holy_order_government } }
		change_government = holy_order_government
	}
	add_realm_law_skip_effects = holy_order_succession_law
	if = {
		limit = { NOT = { has_variable = templar_simulated_order } }
		set_variable = templar_simulated_order
		templar_initialize_order_effect = yes
	}
	if = {
		limit = { NOT = { has_trait = templar_grand_master } }
		add_trait = templar_grand_master
	}
	if = {
		limit = { NOT = { has_trait = order_member } }
		add_trait = order_member
	}
	if = {
		limit = { NOT = { has_character_modifier = founded_holy_order_modifier } }
		add_character_modifier = { modifier = founded_holy_order_modifier }
	}
	if = {
		limit = { NOT = { has_character_modifier = templar_commandery_domain_administration_modifier } }
		add_character_modifier = { modifier = templar_commandery_domain_administration_modifier }
	}
	set_up_templar_chapter_domicile_effect = yes
}

templar_become_grandmaster_effect = {
	scope:recipient = { save_scope_as = templar_old_grandmaster }
	scope:recipient.primary_title = { save_scope_as = templar_ho_title }
	scope:actor = {
		if = {
			limit = { exists = primary_heir }
			primary_heir = { save_scope_as = templar_realm_heir }
		}
		save_scope_as = templar_new_grandmaster
	}

	create_title_and_vassal_change = {
		type = granted
		save_scope_as = templar_ho_change
		add_claim_on_loss = no
	}
	scope:templar_ho_title = {
		change_title_holder = {
			holder = scope:templar_new_grandmaster
			change = scope:templar_ho_change
		}
	}
	resolve_title_and_vassal_change = scope:templar_ho_change

	scope:templar_new_grandmaster = {
		templar_ensure_playable_grandmaster_effect = yes

		create_title_and_vassal_change = {
			type = abdicated
			save_scope_as = templar_strip_change
			add_claim_on_loss = no
		}
		every_held_title = {
			limit = {
				this != scope:templar_ho_title
				is_noble_family_title = no
				tier = scope:templar_new_grandmaster.highest_held_title_tier
			}
			if = {
				limit = { exists = scope:templar_realm_heir }
				change_title_holder_include_vassals = {
					holder = scope:templar_realm_heir
					change = scope:templar_strip_change
				}
			}
			else = {
				change_title_holder_include_vassals = {
					holder = scope:templar_old_grandmaster
					change = scope:templar_strip_change
				}
			}
		}
		every_held_title = {
			limit = {
				this != scope:templar_ho_title
				is_noble_family_title = no
				tier < scope:templar_new_grandmaster.highest_held_title_tier
			}
			if = {
				limit = { exists = scope:templar_realm_heir }
				change_title_holder = {
					holder = scope:templar_realm_heir
					change = scope:templar_strip_change
				}
			}
			else = {
				change_title_holder = {
					holder = scope:templar_old_grandmaster
					change = scope:templar_strip_change
				}
			}
		}
		resolve_title_and_vassal_change = scope:templar_strip_change

		if = {
			limit = { scope:templar_old_grandmaster = { has_variable = templar_simulated_order } }
			set_variable = { name = templar_discipline value = scope:templar_old_grandmaster.var:templar_discipline }
			set_variable = { name = templar_zeal value = scope:templar_old_grandmaster.var:templar_zeal }
			set_variable = { name = templar_reputation value = scope:templar_old_grandmaster.var:templar_reputation }
			set_variable = { name = templar_commanderies value = scope:templar_old_grandmaster.var:templar_commanderies }
			set_variable = { name = templar_rank value = scope:templar_old_grandmaster.var:templar_rank }
			if = {
				limit = { scope:templar_old_grandmaster = { has_variable_list = templar_commanderies_list } }
				scope:templar_old_grandmaster = {
					every_in_list = {
						variable = templar_commanderies_list
						set_variable = { name = templar_commandery_order value = scope:templar_new_grandmaster }
						scope:templar_new_grandmaster = {
							add_to_variable_list = { name = templar_commanderies_list target = prev }
						}
					}
				}
			}
			templar_update_order_effect = yes
			templar_rebuild_commandery_bonuses_effect = yes
		}

		add_piety = 250
		add_prestige = 150
	}

	scope:templar_old_grandmaster = {
		if = {
			limit = {
				is_alive = yes
				is_ruler = no
			}
			scope:templar_new_grandmaster = { add_courtier = scope:templar_old_grandmaster }
		}
	}
}
