templar_initialize_chapter_voices_effect = {
	if = {
		limit = {
			is_ruler = yes
			templar_is_active_grandmaster_trigger = yes
			has_character_flag = templar_playable_grandmaster
			NOT = { has_variable = templar_chapter_voices_initialized }
		}
		set_variable = templar_chapter_voices_initialized
		templar_create_chapter_voice_title_effect = { SLOT = 1 }
		templar_create_chapter_voice_title_effect = { SLOT = 2 }
		templar_create_chapter_voice_title_effect = { SLOT = 3 }
		templar_create_chapter_voice_title_effect = { SLOT = 4 }
		templar_create_chapter_voice_title_effect = { SLOT = 5 }
	}
}

templar_auto_appoint_senior_knights_effect = {
	if = {
		limit = {
			is_ruler = yes
			templar_is_active_grandmaster_trigger = yes
			has_character_flag = templar_playable_grandmaster
		}
		templar_initialize_chapter_voices_effect = yes
		while = {
			limit = {
				any_court_position_holder = {
					type = templar_senior_knight_court_position
					count < 5
				}
				any_courtier = {
					templar_senior_knight_validity_trigger = { EMPLOYER = root }
					NOT = { has_court_position = templar_senior_knight_court_position }
				}
			}
			count = 5
			ordered_courtier = {
				limit = {
					templar_senior_knight_validity_trigger = { EMPLOYER = root }
					NOT = { has_court_position = templar_senior_knight_court_position }
				}
				order_by = {
					value = prowess
					add = martial
				}
				max = 1
				check_range_bounds = no
				save_scope_as = templar_senior_knight_candidate
				if = {
					limit = { NOT = { is_knight_of = liege } }
					set_knight_status = force
				}
			}
			if = {
				limit = { exists = scope:templar_senior_knight_candidate }
				appoint_court_position = {
					recipient = scope:templar_senior_knight_candidate
					court_position = templar_senior_knight_court_position
				}
			}
		}
	}
}

templar_create_chapter_voice_title_effect = {
	save_scope_as = templar_voice_liege
	create_dynamic_title = {
		tier = barony
		name = templar_chapter_voice_title_name
	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = templar_voice_change
		add_claim_on_loss = no
	}
	scope:new_title = {
		set_variable = { name = templar_chapter_voice value = $SLOT$ }
		set_variable = templar_chapter_voice_unassigned
		set_no_automatic_claims = yes
		set_destroy_if_invalid_heir = no
		set_definitive_form = yes
		change_title_holder = {
			holder = scope:templar_voice_liege
			change = scope:templar_voice_change
		}
	}
	resolve_title_and_vassal_change = scope:templar_voice_change
}

templar_grant_chapter_voice_to_senior_knight_effect = {
	scope:employee = { save_scope_as = templar_voice_knight }
	scope:liege = {
		if = {
			limit = {
				any_held_title = {
					has_variable = templar_chapter_voice_unassigned
				}
			}
			ordered_held_title = {
				limit = { has_variable = templar_chapter_voice_unassigned }
				order_by = var:templar_chapter_voice
				save_scope_as = templar_voice_title
			}
		}
		else = {
			templar_create_chapter_voice_title_effect = { SLOT = 99 }
			ordered_held_title = {
				limit = { has_variable = templar_chapter_voice_unassigned }
				order_by = var:templar_chapter_voice
				save_scope_as = templar_voice_title
			}
		}
	}
	if = {
		limit = { exists = scope:templar_voice_title }
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = templar_voice_grant
			add_claim_on_loss = no
		}
		scope:templar_voice_title = {
			remove_variable = templar_chapter_voice_unassigned
			change_title_holder = {
				holder = scope:templar_voice_knight
				change = scope:templar_voice_grant
			}
		}
		resolve_title_and_vassal_change = scope:templar_voice_grant
		scope:templar_voice_knight = {
			set_variable = { name = templar_chapter_voice_title value = scope:templar_voice_title }
		}
	}
}

templar_revoke_chapter_voice_from_senior_knight_effect = {
	scope:employee = {
		if = {
			limit = { exists = var:templar_chapter_voice_title }
			var:templar_chapter_voice_title = { save_scope_as = templar_voice_title }
			create_title_and_vassal_change = {
				type = revoked
				save_scope_as = templar_voice_revoke
				add_claim_on_loss = no
			}
			scope:templar_voice_title = {
				set_variable = templar_chapter_voice_unassigned
				change_title_holder = {
					holder = scope:liege
					change = scope:templar_voice_revoke
				}
			}
			resolve_title_and_vassal_change = scope:templar_voice_revoke
			remove_variable = templar_chapter_voice_title
		}
	}
}
