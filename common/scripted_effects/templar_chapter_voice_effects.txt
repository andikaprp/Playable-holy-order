templar_initialize_chapter_voices_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = templar_chapter_voices_initialized }
		}
		add_character_flag = templar_chapter_voices_initialized
		templar_cleanup_legacy_chapter_voices_effect = yes
	}
}

templar_cleanup_legacy_chapter_voices_effect = {
	every_held_title = {
		limit = {
			OR = {
				has_variable = templar_chapter_voice
				has_variable = templar_chapter_voice_unassigned
			}
		}
		root = { destroy_title = prev }
	}
}

templar_spawn_chapter_brother_effect = {
	save_scope_as = templar_chapter_gm
	# No template: the vanilla 'knight' template reads root, which is unset in
	# on_action contexts and corrupts the created character (invalid culture/faith).
	create_character = {
		employer = scope:templar_chapter_gm
		age = { 22 40 }
		gender = male
		random_traits = yes
		trait = education_martial_3
		martial = { 8 14 }
		prowess = { 10 16 }
		dynasty = scope:templar_chapter_gm.dynasty
		faith = scope:templar_chapter_gm.faith
		culture = scope:templar_chapter_gm.culture
		save_scope_as = templar_spawned_brother
	}
	scope:templar_spawned_brother = {
		set_knight_status = force
	}
}

templar_ensure_chapter_brethren_pool_effect = {
	save_scope_as = templar_chapter_gm
	while = {
		limit = {
			NOR = {
				any_courtier = {
					templar_senior_knight_auto_appoint_trigger = { EMPLOYER = scope:templar_chapter_gm }
					count = 3
				}
			}
		}
		count = 3
		templar_spawn_chapter_brother_effect = yes
	}
}

templar_auto_appoint_senior_knights_effect = {
	if = {
		limit = {
			is_ruler = yes
			templar_is_active_grandmaster_trigger = yes
			any_held_title = { is_holy_order = yes }
		}
		save_scope_as = templar_chapter_gm
		templar_ensure_chapter_brethren_pool_effect = yes
		while = {
			limit = {
				templar_senior_knight_slots_open_trigger = yes
				any_courtier = {
					templar_senior_knight_auto_appoint_trigger = { EMPLOYER = scope:templar_chapter_gm }
					NOT = { has_court_position = templar_senior_knight_court_position }
				}
			}
			count = 3
			ordered_courtier = {
				limit = {
					templar_senior_knight_auto_appoint_trigger = { EMPLOYER = scope:templar_chapter_gm }
					NOT = { has_court_position = templar_senior_knight_court_position }
				}
				order_by = {
					value = prowess
					add = martial
				}
				max = 1
				check_range_bounds = no
				save_scope_as = templar_senior_knight_candidate
				if = {
					limit = { NOT = { is_knight_of = liege } }
					set_knight_status = force
				}
			}
			if = {
				limit = { exists = scope:templar_senior_knight_candidate }
				appoint_court_position = {
					recipient = scope:templar_senior_knight_candidate
					court_position = templar_senior_knight_court_position
				}
				scope:templar_senior_knight_candidate = {
					save_scope_as = employee
					scope:templar_chapter_gm = { save_scope_as = liege }
					templar_register_senior_knight_effect = yes
				}
			}
		}
		templar_repair_senior_knight_election_effect = yes
	}
}

templar_ensure_grandmaster_dynasty_effect = {
	if = {
		limit = { NOT = { exists = dynasty } }
		create_dynasty = {
			name = dynn_de_pagan
		}
	}
}

templar_register_senior_knight_effect = {
	scope:liege = {
		templar_ensure_grandmaster_dynasty_effect = yes
		if = {
			limit = { primary_title = { is_holy_order = yes } }
			primary_title = { save_scope_as = templar_ho_title }
		}
		else = {
			random_held_title = {
				limit = { is_holy_order = yes }
				save_scope_as = templar_ho_title
			}
		}
	}
	scope:employee = {
		add_character_flag = templar_senior_knight
		if = {
			limit = { has_trait = order_member }
			add_character_flag = templar_had_order_member
			remove_trait = order_member
		}
		if = {
			limit = {
				exists = scope:liege.house
				NOT = { house = scope:liege.house }
			}
			set_house = scope:liege.house
		}
		if = {
			limit = {
				exists = scope:templar_ho_title
				NOT = { has_claim_on = scope:templar_ho_title }
			}
			add_pressed_claim = scope:templar_ho_title
		}
	}
}

templar_unregister_senior_knight_effect = {
	scope:employee = {
		remove_character_flag = templar_senior_knight
		if = {
			limit = { has_character_flag = templar_had_order_member }
			remove_character_flag = templar_had_order_member
			add_trait = order_member
		}
	}
}

templar_link_senior_knight_to_order_dynasty_effect = {
	scope:liege = { save_scope_as = liege }
	scope:employee = { save_scope_as = employee }
	templar_register_senior_knight_effect = yes
}

templar_repair_senior_knight_election_effect = {
	templar_ensure_grandmaster_dynasty_effect = yes
	every_court_position_holder = {
		type = templar_senior_knight_court_position
		save_scope_as = employee
		root = { save_scope_as = liege }
		templar_register_senior_knight_effect = yes
	}
}

templar_auto_nominate_chapter_successor_effect = {
	if = {
		limit = {
			is_ruler = yes
			any_held_title = { is_holy_order = yes }
			NOT = { exists = var:templar_chapter_nominee }
			any_court_position_holder = { type = templar_senior_knight_court_position }
		}
		every_court_position_holder = {
			type = templar_senior_knight_court_position
			limit = { NOT = { exists = scope:templar_auto_nominee } }
			save_scope_as = templar_auto_nominee
		}
		if = {
			limit = { exists = scope:templar_auto_nominee }
			scope:templar_auto_nominee = { save_scope_as = templar_nominee }
			set_variable = { name = templar_chapter_nominee value = scope:templar_nominee }
		}
	}
}

	}
}

templar_cleanup_stale_apostolic_voice_effect = {
	every_living_character = {
		limit = {
			has_character_flag = templar_apostolic_voice_holder
			OR = {
				is_alive = no
				root.faith ?= {
					NOT = { religious_head = prev }
				}
			}
		}
		templar_revoke_apostolic_voice_effect = yes
	}
}

templar_ensure_pope_apostolic_elector_effect = {
	if = {
		limit = {
			is_ruler = yes
			any_held_title = { is_holy_order = yes }
			faith = { exists = religious_head }
		}
		templar_cleanup_stale_apostolic_voice_effect = yes
		faith.religious_head = {
			if = {
				limit = {
					is_alive = yes
					is_ruler = yes
					NOT = { this = root }
					NOT = { has_character_flag = templar_apostolic_voice_holder }
				}
				save_scope_as = templar_apostolic_voice_candidate
			}
		}
		if = {
			limit = { exists = scope:templar_apostolic_voice_candidate }
			scope:templar_apostolic_voice_candidate = { save_scope_as = employee }
			save_scope_as = liege
			templar_grant_apostolic_voice_effect = yes
		}
	}
}

templar_grant_apostolic_voice_effect = {
	scope:employee = { save_scope_as = templar_apostolic_voice_holder }
	scope:liege = { save_scope_as = templar_apostolic_voice_liege }
	if = {
		limit = {
			scope:templar_apostolic_voice_holder = {
				NOT = { has_character_flag = templar_apostolic_voice_holder }
			}
		}
		scope:templar_apostolic_voice_liege = {
			if = {
				limit = { has_domicile = yes }
				domicile.domicile_location = { save_scope_as = templar_apostolic_voice_capital_province }
			}
		}
		create_dynamic_title = {
			tier = duchy
			name = templar_apostolic_voice_title_name
		}
		scope:templar_apostolic_voice_liege = {
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = templar_apostolic_voice_change
				add_claim_on_loss = no
			}
		}
		scope:new_title = {
			set_variable = templar_apostolic_voice
			set_landless_title = yes
			set_no_automatic_claims = yes
			set_destroy_if_invalid_heir = no
			set_destroy_on_succession = no
			set_delete_on_destroy = yes
			set_definitive_form = yes
			set_can_be_named_after_dynasty = no
			if = {
				limit = { exists = scope:templar_apostolic_voice_capital_province }
				set_capital_county = scope:templar_apostolic_voice_capital_province.county
			}
			change_title_holder = {
				holder = scope:templar_apostolic_voice_holder
				change = scope:templar_apostolic_voice_change
			}
		}
		resolve_title_and_vassal_change = scope:templar_apostolic_voice_change
		scope:templar_apostolic_voice_holder = {
			if = {
				limit = {
					NOT = { government_has_flag = government_is_feudal }
					NOT = { government_has_flag = government_is_clan }
				}
				change_government = feudal_government
			}
			if = {
				limit = { NOT = { is_vassal_of = scope:templar_apostolic_voice_liege } }
				save_scope_as = templar_voice_knight
				scope:templar_apostolic_voice_liege = { save_scope_as = templar_voice_liege }
				templar_repair_chapter_voice_vassalage_effect = yes
			}
			set_variable = { name = templar_apostolic_voice_title value = scope:new_title }
			add_character_flag = templar_apostolic_voice_holder
		}
	}
}

templar_revoke_apostolic_voice_effect = {
	if = {
		limit = { exists = var:templar_apostolic_voice_title }
		var:templar_apostolic_voice_title = { save_scope_as = templar_apostolic_voice_title }
		if = {
			limit = { exists = liege }
			liege = { destroy_title = scope:templar_apostolic_voice_title }
		}
		remove_variable = templar_apostolic_voice_title
	}
	remove_character_flag = templar_apostolic_voice_holder
}

templar_repair_chapter_voice_vassalage_effect = {
	if = {
		limit = {
			exists = scope:templar_voice_knight
			exists = scope:templar_voice_liege
			scope:templar_voice_knight = {
				is_alive = yes
				NOT = { is_vassal_of = scope:templar_voice_liege }
			}
		}
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = templar_voice_vassal_change
			add_claim_on_loss = no
		}
		scope:templar_voice_knight = {
			change_liege = {
				liege = scope:templar_voice_liege
				change = scope:templar_voice_vassal_change
			}
		}
		resolve_title_and_vassal_change = scope:templar_voice_vassal_change
	}
}
