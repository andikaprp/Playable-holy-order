templar_holy_order_elective = {
	elector_vote_strength = {
		base = 1
		modifier = {
			desc = templar_holy_order_vote_strength_grandmaster
			add = 5
			this = scope:holder
		}
		modifier = {
			desc = templar_holy_order_vote_strength_senior_knight
			add = 2
			templar_holy_order_election_senior_knight_candidate_trigger = yes
		}
		modifier = {
			desc = templar_holy_order_vote_strength_pope
			add = 4
			templar_holy_order_pope_elector_trigger = yes
		}
	}

	electors = {
		add = holder
		add = {
			type = holder_direct_vassals
			limit = {
				is_capable_adult = yes
				is_imprisoned = no
			}
		}
		max = 15
	}

	candidate_score = {
		base = 0
		elector_self_voting_pattern_feudal_elective_modifier = yes
		holder_voting_pattern_feudal_elective_modifier = yes
		elector_voting_pattern_circumstances_feudal_elective_modifier = yes
		elector_voting_pattern_prestige_piety_feudal_elective_modifier = yes
		elector_voting_pattern_traits_feudal_elective_modifier = yes
		elector_voting_pattern_opinion_feudal_elective_modifier = yes
		elector_voting_pattern_circumstances_misc_elective_modifier = yes
		modifier = {
			add = 25
			scope:candidate = {
				templar_holy_order_election_senior_knight_candidate_trigger = yes
			}
		}
		modifier = {
			add = 10000
			scope:holder ?= {
				exists = var:templar_chapter_nominee
				var:templar_chapter_nominee = scope:candidate
			}
		}
	}

	candidates = {
		# Senior Knights are moved into the Grand Master's dynasty when appointed,
		# so holder_dynasty covers the chapter brethren.
		add = {
			type = holder_dynasty
			limit = {
				templar_holy_order_election_senior_knight_candidate_trigger = yes
			}
		}
		# Fallback: anyone holding a pressed claim on the order title.
		add = {
			type = title_claimants
			limit = {
				is_capable_adult = yes
				is_imprisoned = no
				NOT = { has_trait = excommunicated }
			}
		}
		# Last resort: chapter brothers at court, before Senior Knights are appointed.
		add = {
			type = holder_dynasty
			limit = {
				is_capable_adult = yes
				is_imprisoned = no
				NOT = { has_trait = excommunicated }
				is_courtier_of = scope:holder
				NOT = { this = scope:holder }
				faith = scope:holder.faith
			}
		}
	}
}
