templar_order.0002 = {
	type = character_event
	title = templar_order.0002.t
	desc = templar_order.0002.desc
	theme = realm
	left_portrait = root
	immediate = {
		random_in_list = { variable = templar_commanderies_list save_scope_as = templar_commandery_1 }
		if = {
			limit = { variable_list_size = { name = templar_commanderies_list value >= 2 } }
			random_in_list = {
				variable = templar_commanderies_list
				limit = { this != scope:templar_commandery_1 }
				save_scope_as = templar_commandery_2
			}
		}
		if = {
			limit = { variable_list_size = { name = templar_commanderies_list value >= 3 } }
			random_in_list = {
				variable = templar_commanderies_list
				limit = {
					this != scope:templar_commandery_1
					this != scope:templar_commandery_2
				}
				save_scope_as = templar_commandery_3
			}
		}
		if = {
			limit = { variable_list_size = { name = templar_commanderies_list value >= 4 } }
			random_in_list = {
				variable = templar_commanderies_list
				limit = {
					this != scope:templar_commandery_1
					this != scope:templar_commandery_2
					this != scope:templar_commandery_3
				}
				save_scope_as = templar_commandery_4
			}
		}
		if = {
			limit = { variable_list_size = { name = templar_commanderies_list value >= 5 } }
			random_in_list = {
				variable = templar_commanderies_list
				limit = {
					this != scope:templar_commandery_1
					this != scope:templar_commandery_2
					this != scope:templar_commandery_3
					this != scope:templar_commandery_4
				}
				save_scope_as = templar_commandery_5
			}
		}
	}
	option = { name = templar_order.0002.a trigger = { exists = scope:templar_commandery_1 } }
	option = { name = templar_order.0002.b trigger = { exists = scope:templar_commandery_2 } }
	option = { name = templar_order.0002.c trigger = { exists = scope:templar_commandery_3 } }
	option = { name = templar_order.0002.d trigger = { exists = scope:templar_commandery_4 } }
	option = { name = templar_order.0002.e trigger = { exists = scope:templar_commandery_5 } }
	option = { name = templar_order.0002.f }
}

templar_order.0600 = {
	type = character_event
	title = templar_order.0600.t
	desc = templar_order.0600.desc
	theme = realm
	left_portrait = root
	right_portrait = scope:templar_borrower

	trigger = {
		has_variable = templar_temple_bank_active
		exists = scope:templar_borrower
		scope:templar_borrower = { NOT = { exists = var:templar_loan_amount_owed } }
	}

	immediate = {
		if = { limit = { var:templar_rank >= 5 } set_variable = { name = templar_loan_offer_amount value = 900 } }
		else_if = { limit = { var:templar_rank >= 4 } set_variable = { name = templar_loan_offer_amount value = 600 } }
		else_if = { limit = { var:templar_rank >= 3 } set_variable = { name = templar_loan_offer_amount value = 400 } }
		else_if = { limit = { var:templar_rank >= 2 } set_variable = { name = templar_loan_offer_amount value = 250 } }
		else = { set_variable = { name = templar_loan_offer_amount value = 150 } }
	}

	option = {
		name = templar_order.0600.a
		trigger = { gold >= var:templar_loan_offer_amount }
		templar_grant_loan_effect = {
			LENDER = root
			BORROWER = scope:templar_borrower
			AMOUNT = var:templar_loan_offer_amount
		}
	}
	option = {
		name = templar_order.0600.b
		trigger = { gold >= 100 }
		templar_grant_loan_effect = {
			LENDER = root
			BORROWER = scope:templar_borrower
			AMOUNT = 100
		}
	}
	option = {
		name = templar_order.0600.c
	}

	after = {
		remove_variable = templar_loan_offer_amount
	}
}

templar_order.0700 = {
	type = character_event
	title = templar_order.0700.t
	desc = templar_order.0700.desc
	theme = realm
	left_portrait = root

	immediate = {
		if = {
			limit = { has_variable_list = owes_me_money }
			random_in_list = { variable = owes_me_money save_scope_as = templar_debtor_1 }
			if = {
				limit = { variable_list_size = { name = owes_me_money value >= 2 } }
				random_in_list = {
					variable = owes_me_money
					limit = { this != scope:templar_debtor_1 }
					save_scope_as = templar_debtor_2
				}
			}
			if = {
				limit = { variable_list_size = { name = owes_me_money value >= 3 } }
				random_in_list = {
					variable = owes_me_money
					limit = { this != scope:templar_debtor_1 this != scope:templar_debtor_2 }
					save_scope_as = templar_debtor_3
				}
			}
		}
		if = {
			limit = { has_variable_list = templar_depositors }
			random_in_list = { variable = templar_depositors save_scope_as = templar_depositor_1 }
			if = {
				limit = { variable_list_size = { name = templar_depositors value >= 2 } }
				random_in_list = {
					variable = templar_depositors
					limit = { this != scope:templar_depositor_1 }
					save_scope_as = templar_depositor_2
				}
			}
		}
	}

	option = { name = templar_order.0700.a trigger = { exists = scope:templar_debtor_1 } }
	option = { name = templar_order.0700.b trigger = { exists = scope:templar_debtor_2 } }
	option = { name = templar_order.0700.c trigger = { exists = scope:templar_debtor_3 } }
	option = { name = templar_order.0700.d trigger = { exists = scope:templar_depositor_1 } }
	option = { name = templar_order.0700.e trigger = { exists = scope:templar_depositor_2 } }
	option = {
		name = templar_order.0700.f
		add_piety = 25
	}
}

templar_order.2000 = {
	type = character_event
	title = templar_order.2000.t
	desc = templar_order.2000.desc
	theme = realm
	left_portrait = root
	right_portrait = scope:templar_revoking_patron

	trigger = {
		has_variable = templar_simulated_order
		exists = scope:templar_revoked_commandery
		exists = scope:templar_revoking_patron
		scope:templar_revoked_commandery = { var:templar_commandery_order ?= root }
	}

	option = {
		name = templar_order.2000.a
		scope:templar_revoking_patron = {
			add_opinion = { target = root modifier = grateful_opinion opinion = 20 }
		}
		add_piety = 200
		templar_change_reputation_effect = { VALUE = -3 }
		templar_remove_commandery_effect = { COMMANDERY = scope:templar_revoked_commandery }
	}

	option = {
		name = templar_order.2000.b
		trigger = { gold >= 200 prestige >= 300 }
		remove_short_term_gold = 200
		add_prestige = -300
		scope:templar_revoked_commandery = {
			change_variable = { name = templar_commandery_loyalty add = 35 }
		}
		scope:templar_revoking_patron = {
			add_opinion = { target = root modifier = grateful_opinion opinion = 10 }
		}
		templar_change_reputation_effect = { VALUE = 2 }
		templar_rebuild_commandery_bonuses_effect = yes
	}

	option = {
		name = templar_order.2000.c
		scope:templar_revoking_patron = {
			add_opinion = { target = root modifier = angry_opinion opinion = -40 }
		}
		add_dread = 10

			60 = {
